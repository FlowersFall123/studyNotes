// 适用于 @Before, @After 等通知。关键是通过 JoinPoint 获取上下文。 
@Before("execution(* com.example.service.UserService.*(..))")
public void logArgs(JoinPoint joinPoint) {
    // 1. 获取方法签名（方法名）
    String methodName = joinPoint.getSignature().getName();

    // 2. 获取参数列表（数组）
    Object[] args = joinPoint.getArgs();

    System.out.println("正在执行方法: " + methodName);
    System.out.println("参数是: " + Arrays.toString(args));
}
// 如果你想记录方法的返回值，需要使用 returning 属性，其值必须与参数名一致。 
// returning 的值 "result" 必须和方法参数名一致
@AfterReturning(pointcut = "execution(* com.service.UserService.*(..))", returning = "result")
public void logResult(Object result) {
    System.out.println("方法执行成功，返回值为: " + result);
    // 注意：在这里你只能读取返回值，不能修改它。
}
// 最强大的通知。可以修改参数、篡改返回值，甚至阻止目标方法执行。 
@Around("execution(* com.service.UserService.getUser(..))")
public Object auditMethod(ProceedingJoinPoint pjp) throws Throwable {
    Object[] args = pjp.getArgs();
    
    // 【修改参数】：如果第一个参数是 String，把它改成 "ModifiedName"
    if (args.length > 0 && args[0] instanceof String) {
        args[0] = "ModifiedName";
    }

    Object result = null;
    try {
        // 【关键】：执行目标方法。如果不调这行，目标方法就被拦截了！
        result = pjp.proceed(args); 
        
        // 【篡改返回值】：
        if (result instanceof String) {
            result = result + " (被AOP修改了)";
        }
    } catch (Throwable e) {
        throw e; 
    }
    return result;
}
// 生产级实战中，推荐使用自定义注解。 
// 1. 定义注解
@Target(ElementType.METHOD)
@Retention(RetentionPolicy.RUNTIME)
public @interface RecordLog { String value() default ""; }

// 2. 编写切面
@Aspect @Component
public class AnnotationAspect {
    @Around("@annotation(recordLog)") 
    public Object log(ProceedingJoinPoint pjp, RecordLog recordLog) throws Throwable {
        String action = recordLog.value();
        System.out.println("执行操作: " + action);
        return pjp.proceed();
    }
}

// 当无法预知参数类型时，使用 ReflectionUtils 进行通用处理（如统一去空格）。 
@Around("execution(* com.service..*.*(..))")
public Object genericHandle(ProceedingJoinPoint pjp) throws Throwable {
    Object[] args = pjp.getArgs();

    for (Object arg : args) {
        if (arg == null) continue;

        // 场景1：利用反射查找并填充字段
        Field field = ReflectionUtils.findField(arg.getClass(), "operator");
        if (field != null) {
            ReflectionUtils.makeAccessible(field);
            ReflectionUtils.setField(field, arg, "SystemAdmin");
        }

        // 场景2：遍历所有字段去空格
        ReflectionUtils.doWithFields(arg.getClass(), f -> {
            ReflectionUtils.makeAccessible(f);
            if (f.get(arg) instanceof String val) {
                f.set(arg, val.trim());
            }
        });
    }
    return pjp.proceed(args);
}
// 性能优化：让 DTO 实现公共接口，避免反射损耗，类型更安全。 
// 1. 定义公共接口
public interface BaseEntity {
    void setUpdateTime(Date date);
    void setOperator(String operator);
}

// 2. 切面中直接强转 (高性能)
@Around("execution(* com.service..*.*(..))")
public Object interfaceHandle(ProceedingJoinPoint pjp) throws Throwable {
    Object[] args = pjp.getArgs();

    for (Object arg : args) {
        // 只要实现了接口，直接操作，无需反射
        if (arg instanceof BaseEntity) {
            BaseEntity entity = (BaseEntity) arg;
            entity.setUpdateTime(new Date());
            entity.setOperator("SystemAdmin");
        }
    }
    return pjp.proceed(args);
}

避坑指南：自调用失效

90% 的新手都会踩的坑：为什么 this.method() 不触发切面？

Spring AOP 是基于代理模式 (Proxy) 实现的。当你直接调用类内部的方法时，你绕过了 Spring 生成的代理对象，直接操作了原始对象。

外部调用 (External)

Caller -> Proxy -> Aspect -> Target。切面生效。

内部调用 (Internal this)

Method A -> this.Method B。绕过代理，直接调用目标。切面失效。
